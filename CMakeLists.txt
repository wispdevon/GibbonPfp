cmake_minimum_required(VERSION 3.24)
project(GibbonPfp VERSION 0.1.0 LANGUAGES CXX)
set(GIBBON_BUILD_VERSION "${PROJECT_VERSION}" CACHE STRING "Version reported by the installed application")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
add_compile_options("$<$<CXX_COMPILER_ID:MSVC>:/utf-8>")
include(CTest)
include(GNUInstallDirs)
option(GIBBON_SYSTEM_INSTALL "Use system libraries and the Linux desktop install layout" OFF)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Quick QuickControls2 Concurrent Network Test)
find_package(OpenCV REQUIRED COMPONENTS core imgproc)
option(GIBBON_BUNDLED_ORT "Use pinned official ONNX Runtime CPU binaries" OFF)
if(GIBBON_BUNDLED_ORT)
  include(cmake/OnnxRuntime.cmake)
else()
  find_package(onnxruntime CONFIG REQUIRED)
endif()
find_package(libheif CONFIG REQUIRED)
find_package(TIFF REQUIRED)
find_package(WebP CONFIG QUIET)
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
  pkg_check_modules(RAW QUIET IMPORTED_TARGET libraw_r)
  if(NOT TARGET WebP::webp)
    pkg_check_modules(WEBP REQUIRED IMPORTED_TARGET libwebp libwebpdemux)
  endif()
endif()
if(TARGET WebP::webp)
  set(WEBP_TARGETS WebP::webp WebP::webpdemux)
else()
  set(WEBP_TARGETS PkgConfig::WEBP)
endif()
if(TARGET PkgConfig::RAW)
  set(RAW_TARGET PkgConfig::RAW)
else()
  find_package(libraw CONFIG REQUIRED)
  set(RAW_TARGET libraw::raw_r)
endif()
qt_standard_project_setup(REQUIRES 6.8)
add_library(gibbon_core src/settings.cpp src/engine.cpp src/models.cpp src/codecs.cpp)
target_include_directories(gibbon_core PUBLIC src ${OpenCV_INCLUDE_DIRS})
target_link_libraries(gibbon_core PUBLIC Qt6::Core Qt6::Gui Qt6::Network ${OpenCV_LIBS} onnxruntime::onnxruntime ${RAW_TARGET} heif TIFF::TIFF ${WEBP_TARGETS})
target_compile_definitions(gibbon_core PRIVATE GIBBON_SOURCE_MODELS="${CMAKE_CURRENT_SOURCE_DIR}/assets/models")
qt_add_executable(gibbonpfp src/main.cpp src/controller.cpp src/controller.h)
target_compile_definitions(gibbonpfp PRIVATE GIBBON_BUILD_VERSION="${GIBBON_BUILD_VERSION}")
if(WIN32)
  enable_language(RC)
  target_sources(gibbonpfp PRIVATE assets/windows.rc)
  target_include_directories(gibbonpfp PRIVATE assets)
elseif(APPLE)
  target_sources(gibbonpfp PRIVATE assets/gibbonpfp.icns)
  set_source_files_properties(assets/gibbonpfp.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
  set_target_properties(gibbonpfp PROPERTIES MACOSX_BUNDLE_ICON_FILE gibbonpfp.icns)
endif()
if(WIN32 AND GIBBON_BUNDLED_ORT)
  add_custom_command(TARGET gibbonpfp POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:onnxruntime::onnxruntime>" "$<TARGET_FILE_DIR:gibbonpfp>")
endif()
set_target_properties(gibbonpfp PROPERTIES WIN32_EXECUTABLE FALSE MACOSX_BUNDLE TRUE MACOSX_BUNDLE_BUNDLE_NAME GibbonPfp MACOSX_BUNDLE_GUI_IDENTIFIER space.devonlabs.gibbonpfp MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION})
target_link_libraries(gibbonpfp PRIVATE gibbon_core Qt6::Quick Qt6::QuickControls2 Qt6::Concurrent)
qt_import_plugins(gibbonpfp INCLUDE Qt6::QOffscreenIntegrationPlugin)
qt_add_qml_module(gibbonpfp URI Gibbon VERSION 1.0 QML_FILES qml/Main.qml)
file(GLOB FONT_FILES RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" assets/fonts/*.ttf)
qt_add_resources(gibbonpfp fonts PREFIX / FILES ${FONT_FILES})
qt_add_resources(gibbonpfp icon PREFIX / FILES assets/gibbonpfp.png)
install(TARGETS gibbonpfp BUNDLE DESTINATION . RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(GIBBON_SYSTEM_INSTALL)
  if(NOT UNIX OR APPLE)
    message(FATAL_ERROR "GIBBON_SYSTEM_INSTALL is for Linux packages")
  endif()
  install(DIRECTORY assets/models DESTINATION ${CMAKE_INSTALL_DATADIR}/gibbonpfp
    FILES_MATCHING PATTERN "face*.onnx" PATTERN "human*.onnx" PATTERN "*.json" PATTERN "*LICENSE*")
  install(FILES assets/gibbonpfp.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
  install(FILES assets/gibbonpfp.png DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/256x256/apps)
  install(FILES LICENSE THIRD_PARTY_NOTICES.md DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/gibbonpfp)
  install(DIRECTORY assets/fonts/ DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/gibbonpfp/fonts
    FILES_MATCHING PATTERN "*OFL*" PATTERN "*LICENSE*")
else()
if(APPLE)
  set(ASSET_DEST gibbonpfp.app/Contents/Resources)
else()
  set(ASSET_DEST ${CMAKE_INSTALL_BINDIR})
endif()
install(DIRECTORY assets/models DESTINATION ${ASSET_DEST} FILES_MATCHING PATTERN "face*.onnx" PATTERN "human*.onnx" PATTERN "*.json" PATTERN "*LICENSE*")
install(FILES LICENSE THIRD_PARTY_NOTICES.md DESTINATION ${ASSET_DEST})
install(DIRECTORY assets/licenses/ DESTINATION ${ASSET_DEST}/licenses/qt)
if(DEFINED VCPKG_INSTALLED_DIR)
  install(DIRECTORY "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/share/" DESTINATION ${ASSET_DEST}/licenses/dependencies FILES_MATCHING PATTERN "copyright")
endif()
if(DEFINED ENV{QT_ROOT_DIR})
  file(TO_CMAKE_PATH "$ENV{QT_ROOT_DIR}" QT_LICENSE_ROOT)
  install(DIRECTORY "${QT_LICENSE_ROOT}/../../Licenses/" DESTINATION ${ASSET_DEST}/licenses/qt OPTIONAL)
endif()
install(DIRECTORY assets/fonts/ DESTINATION ${ASSET_DEST}/licenses/fonts FILES_MATCHING PATTERN "*OFL*" PATTERN "*LICENSE*")
if(APPLE)
  set(OFFSCREEN_DEST gibbonpfp.app/Contents/PlugIns/platforms)
else()
  set(OFFSCREEN_DEST ${QT6_INSTALL_PLUGINS}/platforms)
endif()
# Qt deployment normally selects only the desktop platform plugin. GUI smoke
# tests also need offscreen, even though the CLI itself uses QCoreApplication.
install(FILES "$<TARGET_FILE:Qt6::QOffscreenIntegrationPlugin>" DESTINATION ${OFFSCREEN_DEST})
if(APPLE)
  # Installation removes build-tree rpaths before macdeployqt runs. Stage the
  # pinned native runtime first so its @rpath dependencies remain resolvable.
  if(DEFINED VCPKG_INSTALLED_DIR)
    install(DIRECTORY "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/lib/"
      DESTINATION gibbonpfp.app/Contents/Frameworks FILES_MATCHING PATTERN "*.dylib")
  endif()
  if(GIBBON_BUNDLED_ORT)
    install(DIRECTORY "${ort_SOURCE_DIR}/lib/"
      DESTINATION gibbonpfp.app/Contents/Frameworks FILES_MATCHING PATTERN "*.dylib")
  endif()
endif()
if(UNIX AND NOT APPLE)
  # Also deploy native image libraries when Qt comes from a system prefix.
  set(DEPLOY_EXTRA
    PRE_EXCLUDE_REGEXES "^libc[.]so.*" "^libm[.]so.*" "^libpthread[.]so.*" "^libdl[.]so.*" "^librt[.]so.*" "^libutil[.]so.*" "^libresolv[.]so.*" "^libnss_.*" "^ld-linux.*"
    POST_INCLUDE_REGEXES ".*/libQt6.*" ".*/libopencv.*" ".*/libraw.*" ".*/libheif.*" ".*/libde265.*" ".*/libwebp.*" ".*/libtiff.*"
    POST_EXCLUDE_REGEXES "^/usr/lib/" "^/lib/")
endif()
qt_generate_deploy_qml_app_script(TARGET gibbonpfp OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR ${DEPLOY_EXTRA})
install(SCRIPT ${deploy_script})
if(WIN32)
  set(CMAKE_INSTALL_OPENMP_LIBRARIES TRUE)
  set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION bin)
  include(InstallRequiredSystemLibraries)
  install(CODE "file(GLOB runtime_dlls \"${CMAKE_CURRENT_BINARY_DIR}/*.dll\")\nfile(INSTALL DESTINATION \"\${CMAKE_INSTALL_PREFIX}/bin\" TYPE FILE FILES \${runtime_dlls})")
endif()
endif()
if(BUILD_TESTING)
  qt_add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE gibbon_core Qt6::Test)
  add_test(NAME core COMMAND core_tests)
  set_tests_properties(core PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  qt_add_executable(controller_tests tests/controller_tests.cpp src/controller.cpp src/controller.h)
  target_link_libraries(controller_tests PRIVATE gibbon_core Qt6::Quick Qt6::QuickControls2 Qt6::Concurrent Qt6::Test)
  target_compile_definitions(controller_tests PRIVATE GIBBON_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME controller COMMAND controller_tests)
  set_tests_properties(controller PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
endif()
set(CPACK_PACKAGE_NAME GibbonPfp)
set(CPACK_PACKAGE_VENDOR "Devon Labs")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_CONTACT "https://devonlabs.space")
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE")
if(WIN32)
  set(CPACK_GENERATOR "ZIP;NSIS")
  set(CPACK_NSIS_MODIFY_PATH ON)
  set(CPACK_NSIS_MENU_LINKS "bin/gibbonpfp.exe" "GibbonPfp")
elseif(APPLE)
  set(CPACK_GENERATOR "DragNDrop")
else()
  set(CPACK_GENERATOR "TGZ")
endif()
include(CPack)
