cmake_minimum_required(VERSION 3.24)
project(HeadshotFlow VERSION 2.0.1 LANGUAGES CXX)
set(HEADSHOTFLOW_BUILD_VERSION "${PROJECT_VERSION}" CACHE STRING "Version reported by the installed application")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
add_compile_options("$<$<CXX_COMPILER_ID:MSVC>:/utf-8>")
include(CTest)
include(GNUInstallDirs)
option(HEADSHOTFLOW_SYSTEM_INSTALL "Use system libraries and the Linux desktop install layout" OFF)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Quick QuickControls2 Concurrent Network Test)
find_package(OpenCV REQUIRED COMPONENTS core imgproc)
option(HEADSHOTFLOW_BUNDLED_ORT "Use pinned official ONNX Runtime CPU binaries" OFF)
if(HEADSHOTFLOW_BUNDLED_ORT)
  include(cmake/OnnxRuntime.cmake)
else()
  find_package(onnxruntime CONFIG REQUIRED)
endif()
find_package(libheif CONFIG REQUIRED)
find_package(TIFF REQUIRED)
find_package(LibArchive REQUIRED)
find_package(WebP CONFIG QUIET)
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
  pkg_check_modules(RAW QUIET IMPORTED_TARGET libraw_r)
  if(NOT TARGET WebP::webp)
    pkg_check_modules(WEBP REQUIRED IMPORTED_TARGET libwebp libwebpdemux)
  endif()
endif()
if(TARGET WebP::webp)
  set(WEBP_TARGETS WebP::webp WebP::webpdemux)
else()
  set(WEBP_TARGETS PkgConfig::WEBP)
endif()
if(TARGET PkgConfig::RAW)
  set(RAW_TARGET PkgConfig::RAW)
else()
  find_package(libraw CONFIG REQUIRED)
  set(RAW_TARGET libraw::raw_r)
endif()
qt_standard_project_setup(REQUIRES 6.8)
add_library(headshotflow_core src/zip_export.cpp src/settings.cpp src/engine.cpp src/models.cpp src/codecs.cpp)
qt_add_resources(headshotflow_core sample_portraits PREFIX / FILES
  assets/headshotflow-sample.jpg assets/headshotflow-sample-2.jpg assets/headshotflow-sample-3.jpg)
target_include_directories(headshotflow_core PUBLIC src ${OpenCV_INCLUDE_DIRS})
target_link_libraries(headshotflow_core PUBLIC Qt6::Core Qt6::Gui Qt6::Network ${OpenCV_LIBS} onnxruntime::onnxruntime ${RAW_TARGET} heif LibArchive::LibArchive TIFF::TIFF ${WEBP_TARGETS})
target_compile_definitions(headshotflow_core PRIVATE HEADSHOTFLOW_SOURCE_MODELS="${CMAKE_CURRENT_SOURCE_DIR}/assets/models")
qt_add_executable(headshotflow src/main.cpp src/controller.cpp src/recovery.cpp src/controller.h)
target_compile_definitions(headshotflow PRIVATE HEADSHOTFLOW_BUILD_VERSION="${HEADSHOTFLOW_BUILD_VERSION}")
if(WIN32)
  enable_language(RC)
  target_sources(headshotflow PRIVATE assets/windows.rc)
  target_include_directories(headshotflow PRIVATE assets)
elseif(APPLE)
  target_sources(headshotflow PRIVATE assets/headshotflow.icns)
  set_source_files_properties(assets/headshotflow.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
  set_target_properties(headshotflow PROPERTIES MACOSX_BUNDLE_ICON_FILE headshotflow.icns)
endif()
if(WIN32 AND HEADSHOTFLOW_BUNDLED_ORT)
  add_custom_command(TARGET headshotflow POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:onnxruntime::onnxruntime>" "$<TARGET_FILE_DIR:headshotflow>")
endif()
set_target_properties(headshotflow PROPERTIES WIN32_EXECUTABLE FALSE MACOSX_BUNDLE TRUE MACOSX_BUNDLE_BUNDLE_NAME HeadshotFlow MACOSX_BUNDLE_GUI_IDENTIFIER space.devonlabs.headshotflow MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION})
target_link_libraries(headshotflow PRIVATE headshotflow_core Qt6::Quick Qt6::QuickControls2 Qt6::Concurrent)
qt_import_plugins(headshotflow INCLUDE Qt6::QOffscreenIntegrationPlugin)
qt_add_qml_module(headshotflow URI HeadshotFlow VERSION 1.0 QML_FILES qml/Main.qml qml/ExportDialog.qml qml/WorkspaceChoice.qml)
file(GLOB FONT_FILES RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" assets/fonts/*.ttf)
qt_add_resources(headshotflow fonts PREFIX / FILES ${FONT_FILES})
qt_add_resources(headshotflow icon PREFIX / FILES assets/headshotflow.png)
install(TARGETS headshotflow BUNDLE DESTINATION . RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(HEADSHOTFLOW_SYSTEM_INSTALL)
  if(NOT UNIX OR APPLE)
    message(FATAL_ERROR "HEADSHOTFLOW_SYSTEM_INSTALL is for Linux packages")
  endif()
  install(DIRECTORY assets/models DESTINATION ${CMAKE_INSTALL_DATADIR}/headshotflow
    FILES_MATCHING PATTERN "face*.onnx" PATTERN "modnet*.onnx" PATTERN "*.json" PATTERN "*LICENSE*")
  install(FILES assets/headshotflow.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
  install(FILES assets/headshotflow.png DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/256x256/apps)
  install(FILES LICENSE THIRD_PARTY_NOTICES.md DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/headshotflow)
  install(DIRECTORY assets/fonts/ DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/headshotflow/fonts
    FILES_MATCHING PATTERN "*OFL*" PATTERN "*LICENSE*")
else()
if(APPLE)
  set(ASSET_DEST headshotflow.app/Contents/Resources)
else()
  set(ASSET_DEST ${CMAKE_INSTALL_BINDIR})
endif()
install(DIRECTORY assets/models DESTINATION ${ASSET_DEST} FILES_MATCHING PATTERN "face*.onnx" PATTERN "modnet*.onnx" PATTERN "*.json" PATTERN "*LICENSE*")
install(FILES LICENSE THIRD_PARTY_NOTICES.md DESTINATION ${ASSET_DEST})
install(DIRECTORY assets/licenses/ DESTINATION ${ASSET_DEST}/licenses/qt)
if(DEFINED VCPKG_INSTALLED_DIR)
  install(DIRECTORY "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/share/" DESTINATION ${ASSET_DEST}/licenses/dependencies FILES_MATCHING PATTERN "copyright")
endif()
if(DEFINED ENV{QT_ROOT_DIR})
  file(TO_CMAKE_PATH "$ENV{QT_ROOT_DIR}" QT_LICENSE_ROOT)
  install(DIRECTORY "${QT_LICENSE_ROOT}/../../Licenses/" DESTINATION ${ASSET_DEST}/licenses/qt OPTIONAL)
endif()
install(DIRECTORY assets/fonts/ DESTINATION ${ASSET_DEST}/licenses/fonts FILES_MATCHING PATTERN "*OFL*" PATTERN "*LICENSE*")
if(APPLE)
  set(OFFSCREEN_DEST headshotflow.app/Contents/PlugIns/platforms)
else()
  set(OFFSCREEN_DEST ${QT6_INSTALL_PLUGINS}/platforms)
endif()
# Qt deployment normally selects only the desktop platform plugin. GUI smoke
# tests also need offscreen, even though the CLI itself uses QCoreApplication.
install(FILES "$<TARGET_FILE:Qt6::QOffscreenIntegrationPlugin>" DESTINATION ${OFFSCREEN_DEST})
if(APPLE)
  # Installation removes build-tree rpaths before macdeployqt runs. Stage the
  # pinned native runtime first so its @rpath dependencies remain resolvable.
  if(DEFINED VCPKG_INSTALLED_DIR)
    install(DIRECTORY "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/lib/"
      DESTINATION headshotflow.app/Contents/Frameworks FILES_MATCHING PATTERN "*.dylib")
  endif()
  if(HEADSHOTFLOW_BUNDLED_ORT)
    install(FILES "$<TARGET_FILE:onnxruntime::onnxruntime>"
      DESTINATION headshotflow.app/Contents/Frameworks)
  endif()
endif()
if(UNIX AND NOT APPLE)
  # Also deploy native image libraries when Qt comes from a system prefix.
  set(DEPLOY_EXTRA
    PRE_EXCLUDE_REGEXES "^libc[.]so.*" "^libm[.]so.*" "^libpthread[.]so.*" "^libdl[.]so.*" "^librt[.]so.*" "^libutil[.]so.*" "^libresolv[.]so.*" "^libnss_.*" "^ld-linux.*"
    POST_INCLUDE_REGEXES ".*/libQt6.*" ".*/libopencv.*" ".*/libraw.*" ".*/libheif.*" ".*/libde265.*" ".*/libwebp.*" ".*/libtiff.*"
    POST_EXCLUDE_REGEXES "^/usr/lib/" "^/lib/")
endif()
qt_generate_deploy_qml_app_script(TARGET headshotflow OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR ${DEPLOY_EXTRA})
install(SCRIPT ${deploy_script})
if(WIN32)
  set(CMAKE_INSTALL_OPENMP_LIBRARIES TRUE)
  set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION bin)
  include(InstallRequiredSystemLibraries)
  install(CODE "file(GLOB runtime_dlls \"${CMAKE_CURRENT_BINARY_DIR}/*.dll\")\nfile(INSTALL DESTINATION \"\${CMAKE_INSTALL_PREFIX}/bin\" TYPE FILE FILES \${runtime_dlls})")
endif()
endif()
if(BUILD_TESTING)
  qt_add_executable(headshotflow_benchmark benchmarks/runner.cpp)
  target_link_libraries(headshotflow_benchmark PRIVATE headshotflow_core)
  target_compile_definitions(headshotflow_benchmark PRIVATE HEADSHOTFLOW_BENCH_VERSION="${HEADSHOTFLOW_BUILD_VERSION}")
  qt_add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE headshotflow_core Qt6::Test)
  add_test(NAME core COMMAND core_tests -o core-results.txt,txt -o -,txt)
  set_tests_properties(core PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  qt_add_executable(controller_tests tests/controller_tests.cpp src/controller.cpp src/recovery.cpp src/controller.h)
  target_link_libraries(controller_tests PRIVATE headshotflow_core Qt6::Quick Qt6::QuickControls2 Qt6::Concurrent Qt6::Test)
  target_compile_definitions(controller_tests PRIVATE HEADSHOTFLOW_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  # Keep QtTest assertions and benchmark diagnostics visible on Windows.
  set_target_properties(core_tests controller_tests headshotflow_benchmark PROPERTIES WIN32_EXECUTABLE FALSE)
  add_test(NAME controller COMMAND controller_tests -o controller-results.txt,txt -o -,txt)
  set_tests_properties(controller PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
  qt_add_executable(export_tests tests/export_tests.cpp src/controller.cpp src/recovery.cpp src/controller.h)
  target_link_libraries(export_tests PRIVATE headshotflow_core Qt6::Quick Qt6::QuickControls2 Qt6::Concurrent Qt6::Test)
  target_compile_definitions(export_tests PRIVATE HEADSHOTFLOW_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  set_target_properties(export_tests PROPERTIES WIN32_EXECUTABLE FALSE)
  add_test(NAME export COMMAND export_tests -o export-results.txt,txt -o -,txt)
  set_tests_properties(export PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
endif()
set(CPACK_PACKAGE_NAME HeadshotFlow)
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "HeadshotFlow offline headshot editor and batch processor")
set(CPACK_PACKAGE_VENDOR "Devon Labs")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_CONTACT "https://devonlabs.space")
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE")
if(WIN32)
  set(CPACK_GENERATOR "ZIP;NSIS")
  set(CPACK_NSIS_MODIFY_PATH ON)
  set(CPACK_NSIS_MENU_LINKS "bin/headshotflow.exe" "HeadshotFlow")
elseif(APPLE)
  set(CPACK_GENERATOR "DragNDrop")
else()
  set(CPACK_GENERATOR "TGZ")
endif()
include(CPack)
